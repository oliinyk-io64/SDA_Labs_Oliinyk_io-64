#include <stdio.h>
#define MAX_SIZE 5 

int arr[MAX_SIZE]; 
int head = 0;             
int tail = 0;            
int count = 0;            

void enqueue(int pages) {
    if (count == MAX_SIZE) {
        printf("Черга переповнена! Неможливо додати завдання на %d стор.\n", pages);
        return;
    }

    arr[tail] = pages;      
    tail = (tail + 1) % MAX_SIZE;  
    count++;                      
}

int dequeue() {
    if (count == 0) {
        printf("Черга порожня! Немає завдань для друку.\n");
        return -1; 
    }

    int pages = arr[head]; 
    head = (head + 1) % MAX_SIZE;  
    count--;                      
    return pages;
}

int main() {

    enqueue(5); enqueue(3); enqueue(4); enqueue(2); enqueue(1);

    int total_time = 0;
    int taskNum = 1;

    while (count > 0) {
        int pages = dequeue();
        if (pages == -1) break;
        int print_time = pages * 2; 
        total_time += print_time;

        printf("Друкується завдання %d (%d стор.) -> Час: %d сек.\n", 
               taskNum++, pages, print_time);
    }
    printf("\nЗагальний час роботи принтера: %d сек.\n", total_time);
    return 0;
}
