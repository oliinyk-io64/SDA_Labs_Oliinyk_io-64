#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int pages;
    struct Node* next; 
} Node;

Node *head = NULL, *tail = NULL;   
int count = 0;         

void enqueue(int pages) {
    Node* newNode = (Node*) malloc(sizeof(Node));
    newNode->pages = pages;
    newNode->next = NULL; 
    
    if (head == NULL) head = tail = newNode; 
    else { tail->next = newNode; tail = newNode; }
    
    count++;                 
}

int dequeue() {
    if (count == 0) {
        printf("Черга порожня! Немає завдань для друку.\n");
        return -1; 
    }

    Node* temp = head;
    int pages = temp->pages;
    
    head = head->next;
    if (head == NULL) tail = NULL; 
    
    free(temp); 
    count--;                   
    return pages;       
}

int main() {

    enqueue(5); enqueue(3); enqueue(4); enqueue(2); enqueue(1); 

    int total_time = 0, taskNum = 1;

    while (count > 0) {
        int pages = dequeue();
        if (pages == -1) break;

        int printTime = pages * 2; 
        total_time += printTime;
        printf("Друкується завдання %d (%d стор.) Час друку: %d сек.\n", taskNum++, pages, printTime);
    }
    printf("\nЗагальний час роботи принтера: %d сек.\n", total_time);
    return 0;
}
