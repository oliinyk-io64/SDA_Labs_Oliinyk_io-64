#include <stdio.h>
#define _USE_MATH_DEFINES
#include <math.h>

int main() { 
    int n;
    scanf("%d", &n); 

    double x = 1.0 / 3.0;
    double p = sin(x)/x;
    printf("Точний результат P = %.7f\n", p);

    double result = 1;
    int operCount = 0;
    double sum = 0;

    for(int i = 1; i <= n; i++){

        sum += 2*i - 1;
        operCount += 1;

        double res = 1 - ((x*x)/(M_PI*M_PI*sum));
        result *= res;
        operCount += 1;
    };

    printf("Результат за формулою P = %.7f\n", result);
    printf("Кількість операцій: %i\n", operCount);

    return 0;
};
