#include <stdio.h>

int main() {
    //ВВЕДЕННЯ ШУКАНОГО ЧИСЛА
    double x = 3.0;
    printf("Шукане число в рядку: x =%4.1lf\n\n",x);

    double y = 20.0;
    printf("Шукане число в стовпчику: y =%4.1lf\n\n",y);

    //СТВОРЕННЯ ТА ВИВЕДЕННЯ МАТРИЦІ
    int m = 10;
    int n = 10;
    double array[10][10] = {
        {1.0, 2.0, 3.0, 3.0, 3.0, 4.0, 6.0, 8.0, 8.0, 10.0},
        {5.0,  0.0, 4.0,  3.0, 9.0, 1.0, 8.0, 8.0, 2.0, 10.0},
        {2.0,  8.0, 1.0,  7.0, 4.0, 5.0, 6.0, 9.0, 0.0, 12.0},
        {0.0, -1.0, 3.0,  9.0, 8.0, 4.0, 2.0, 9.0, 3.0, 12.0},
        {5.0,  6.0, 7.0,  8.0, 9.0, 1.0, 3.0, 1.0, 4.0, 15.0},
        {4.0,  3.0, 2.0,  1.0, 0.0, 7.0, 8.0, 2.0, 5.0, 15.0},
        {9.0,  2.0, 1.0,  4.0, 5.0, 6.0, 7.0, 3.0, 6.0, 15.0},
        {1.0,  1.0, 2.0,  2.0, 3.0, 3.0, 4.0, 4.0, 7.0, 18.0},
        {0.0,  5.0, 2.0,  9.0, 1.0, 8.0, 4.0, 6.0, -2.0, 20.0},
        {2.0,  2.0, 4.0,  5.0, 6.0, 7.0, 8.0, 9.0, 9.0, 20.0}
    };
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            printf("%4.1f ", array[i][j]);
        }
        printf("\n");
    }

    // ПОШУК У ПЕРШОМУ РЯДКУ
    int left = 0;
    int right = n;

    while(left < right){
        int mid = left + (right - left) / 2;
        if(x <= array[0][mid]){
            right = mid;
        } else{
            left = mid + 1;
        }
    };
    if (left < n && array[0][left] == x) {
        printf("Найлівіше число %.1f знайдено! Координати: рядок 1, стовпчик %d \n\n", x, left + 1);
    } else {
        printf("Число %.1f у першому рядку не знайдено.\n\n", x);
    }
    // ПОШУК В ОСТАННЬОМУ СТОВПЧИКУ

    int top = 0;
    int down = m;

    while(top < down){
        int mid = top + (down - top) / 2;
        if(y <= array[mid][n - 1]){
            down = mid;
        } else{
            top = mid + 1;
        }
    };
    if (top < m && array[top][n - 1] == y) {
        printf("Найлівіше число %.1f знайдено! Координати: рядок %d, стовпчик 10 \n\n", y, top + 1);
    } else {
        printf("Число %.1f у останньому стовпчику не знайдено.\n\n", y);
    }
    return 0;
}
